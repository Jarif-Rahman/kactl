\section{Network flow}
	\subsection{Advanced Flow Models}
		\textbf{Flow with lower bounds:} For edge $u\to v$ with $l\le f\le r$, replace its capacity by $r-l$. Define $b(v)=\sum_{\mathrm{in}}l-\sum_{\mathrm{out}}l$. Add super-source $S$ and super-sink $T$: if $b(v)>0$, add $S\to v$ with capacity $b(v)$; otherwise add $v\to T$ with capacity $-b(v)$. Feasible iff all edges leaving $S$ are saturated. For bounded $s$-$t$ flow, also add $t\to s$ with infinite capacity before checking feasibility.\\
		\textbf{Minimum cost circulation:} Use the lower-bound reduction with edge cost preserved; run min-cost flow from $S$ to $T$. Add $\sum_e l_ec_e$ to the result. For unrestricted circulation, negative-cost residual cycles must also be accounted for.\\
		\textbf{Maximum density induced subgraph:} Maximize $|E(S)|/|S|$. Binary search density $d$. Let $m=|E|$; add $s\to v$ with capacity $m$, $v\to t$ with capacity $m+2d-\deg(v)$, and both directed edges of capacity $1$ for every undirected edge. Density $>d$ exists iff min-cut $<mn$ (for nonempty $S$).\\
		\textbf{Minimum vertex cover (bipartite):} After maximum matching, DFS from unmatched left vertices via unmatched left-to-right edges and matched right-to-left edges. If $Z$ is the set reached, the minimum vertex cover is $(L\setminus Z)\cup(R\cap Z)$.

\section{Matching}
	\subsection{Matching Basics}
		\textbf{Maximum matching:} Repeatedly find augmenting paths in a bipartite graph until none exist.\\
		\textbf{Hall's Theorem:} A bipartite graph $(X,Y)$ has a matching covering all vertices in $X$ iff $|N(S)|\ge |S|$ for every $S\subseteq X$.\\
		\textbf{Kőnig's Theorem:} In a bipartite graph, maximum matching size equals minimum vertex cover size. Maximum independent set size is $|V|$ minus minimum vertex cover size.\\
		\textbf{Node-disjoint path cover (DAG):} Split each vertex into left and right copies; add edge $u_L\to v_R$ for every original edge $u\to v$. Minimum path cover $=n-$ maximum matching.\\
		\textbf{General path cover (DAG):} Use the same matching construction, but add $u_L\to v_R$ whenever $v$ is reachable from $u$ (transitive closure).\\
		\textbf{Dilworth's Theorem:} In a DAG, minimum general path cover size equals maximum antichain size (vertices pairwise unreachable from each other).

\section{Math}
	\subsection{Number of Spanning Trees}
		% I.e. matrix-tree theorem.
		% Source: https://en.wikipedia.org/wiki/Kirchhoff%27s_theorem
		% Test: stress-tests/graph/matrix-tree.cpp
		Create an $N\times N$ matrix \texttt{mat}, and for each edge $a \rightarrow b \in G$, do
		\texttt{mat[a][b]--, mat[b][b]++} (and \texttt{mat[b][a]--, mat[a][a]++} if $G$ is undirected).
		Remove the $i$th row and column and take the determinant; this yields the number of directed spanning trees rooted at $i$
		(if $G$ is undirected, remove any row/column).

	\subsection{Erdős–Gallai theorem}
		% Source: https://en.wikipedia.org/wiki/Erd%C5%91s%E2%80%93Gallai_theorem
		% Test: stress-tests/graph/erdos-gallai.cpp
		A simple graph with node degrees $d_1 \ge \dots \ge d_n$ exists iff $d_1 + \dots + d_n$ is even and for every $k = 1\dots n$,
		\[ \sum _{i=1}^{k}d_{i}\leq k(k-1)+\sum _{i=k+1}^{n}\min(d_{i},k). \]
